	\section{Introduction}

	\begin{frame}{Introduction to Domain Generalization}
		\begin{block}{The Core Objective}
			Train a parametric model $f(\cdot;\theta)$ on a set of $N$ available \textbf{source domains} $\mathcal{S} = \{S_1, S_2, \dots, S_N\}$ such that it generalizes robustly to $K$ completely \textbf{unseen target domains} $\mathcal{T} = \{T_1, T_2, \dots, T_K\}$.
		\end{block}
		\begin{alertblock}{Why Is This Hard?}
			\begin{itemize}
				\item \alert{Zero access} to target-domain data during training no direct supervision signal.
				\item Models must learn \textbf{domain-invariant} features from source data alone.
				\item Distribution shift at test time causes catastrophic accuracy drops in naive ERM.
			\end{itemize}
		\end{alertblock}
	\end{frame}

	\begin{frame}{Related Work: Where Existing Methods Fall Short}
		\begin{columns}[T]
			\begin{column}{0.50\textwidth}
				\begin{block}{Representative DG Approaches}
					\begin{itemize}
						\item \textbf{ERM:} Aggregates all source data learns domain-specific shortcuts.
						\item \textbf{CORAL / MMD:} Aligns first/second-order statistics \alert{single layer only}, loses geometric detail.
						\item \textbf{RSC:} Drops dominant features at training ignores multi-scale structure.
						\item \textbf{Mixup:} Interpolates images across domains does not enforce feature invariance.
					\end{itemize}
				\end{block}
			\end{column}
			\begin{column}{0.50\textwidth}
				\begin{alertblock}{What $M^2$-CL Contributes}
					\begin{itemize}
						\item \textbf{Multi-scale \& multi-layer} extraction captures geometry at every depth.
						\item \textbf{Spatial Dropout} in each block reduces reliance on style shortcuts.
						\item \textbf{Layer-wise contrastive loss} encourages same-class alignment across domains at \emph{all} scales simultaneously.
					\end{itemize}
				\end{alertblock}
			\end{column}
		\end{columns}
	\end{frame}

\begin{frame}{Why ERM Fails: Shortcut Learning}
		\begin{columns}[T]

	\section{The $M^2 - CL$ Architecture}

	\begin{frame}{The $M^2$ Architectural Paradigm}
		\begin{block}{Framework Concept}
			Leverages parallel multi-scale concentration blocks mapped across multiple intermediate layers of a standard backbone network.
		\end{block}
		\begin{figure}
			\centering
			% Đảm bảo file ảnh tồn tại trong thư mục của bạn hoặc giữ nguyên cấu trúc này
			\includegraphics[width=0.75\linewidth]{photos/M2.png}
			% \caption{Visualization of the $M^2$ framework overlaid on a ResNet-18 backbone.}
			\label{fig:m2_architecture}
		\end{figure}
	\end{frame}

	\begin{frame}{Extraction Block}
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\begin{figure}
					\centering
					\includegraphics[width=0.9\linewidth]{photos/extraction.png}
					\caption{Concentration pipeline layout.}
				\end{figure}
			\end{column}
			\begin{column}{0.52\textwidth}
				\begin{block}{Internal Sequence Mechanics}
					\begin{enumerate}
						\item \textbf{$1\times1$ Convolutional Layer:} Compresses high-dimensional channels $c$ down to $\lfloor c/r \rfloor$ using a reduction factor $r$.
						\item \textbf{Spatial Dropout Layer:} Mitigates pixel correlation leakage.
						\item \textbf{Multi-Scale Max Pooling:} Captures spatial features before reduction.
						\item \textbf{MLP Projection Vector:} Maps pooled features into vectors via MLPs.
					\end{enumerate}
				\end{block}
			\end{column}
		\end{columns}
	\end{frame}

\begin{frame}{Spatial Dropout}
	\begin{columns}[T]
		\begin{column}{0.42\textwidth}
			\begin{block}{Vanilla Dropout}
				\small Drops \textbf{individual pixels} --- adjacent pixels remain correlated; style information \alert{leaks through}.
			\end{block}
			\vspace{0.15cm}
			\begin{alertblock}{Spatial Dropout}
				\small Drops \textbf{entire feature maps} ($H \times W$ channels). Encourages the network to rely on alternative feature maps, \textbf{reducing style shortcut reliance} at the source.
			\end{alertblock}
		\end{column}
		\begin{column}{0.58\textwidth}
			\centering
			\includegraphics[width=\linewidth]{photos/spatial_dropout.png}
		\end{column}
	\end{columns}
\end{frame}
\begin{frame}{Multi-Scale Concentration \& Hypercolumn Construction}
		\begin{columns}[T]
			% --- CỘT TRÁI: TEXT TINH GỌN (45% Chiều rộng) ---
			\begin{column}{0.45\textwidth}
				\begin{block}{Spatial Scaling Allocations}
					
					\begin{itemize}
						\item \textbf{Early Hooks:} Pooling sizes optimized for $8\times8$, $4\times4$, and $2\times2$.
						\item \textbf{Late Hooks:} Deeper fields adapt to $7\times7$ and $3\times3$ dimensions.
					\end{itemize}
				\end{block}
				
				\begin{alertblock}{Concatenation Protocol}
					Flattened multi-scale vectors are concatenated into a \textbf{hypercolumn vector} for final FC layer classification.
				\end{alertblock}
			\end{column}
			
			% --- CỘT PHẢI: CHÈN ẢNH MINH HỌA (55% Chiều rộng) ---

	\begin{frame}{The M2-CL Loss Framework}
    \begin{columns}[t]
        % COLUMN 1: STRATEGY & EFFICIENCY
        \begin{column}{0.44\textwidth}
            \begin{block}{Geometric Constraints}
                \begin{itemize}
                    \item \textbf{Pull:} Same class, cross domains.
                    \item \textbf{Push:} Different classes, same style.
                \end{itemize}
            \end{block}
            
            \begin{exampleblock}{Matrix Parallelization}
                \begin{itemize}
                    \item Reduces memory bottlenecks.
                    \item Stacks batch: $U^{(l)T}U^{(l)}$.
                    \item Denominator computed \textbf{once}.
                \end{itemize}
            \end{exampleblock}
        \end{column}

        % COLUMN 2: MATHEMATICS
        \begin{column}{0.53\textwidth}
            \begin{alertblock}{Alignment Measure ($\|u\|=1$)}
                \small
                \[ p^{(l)}(c) = \frac{\sum_{i, j \in N_c} \exp(u_i^{T}u_j / \tau)}{\sum_{i, a \in N_b} \exp(u_i^{T}u_a / \tau)} \]
            \end{alertblock}

            \begin{block}{Total Objective}
                \small
                \[ \mathcal{L} = \mathcal{L}_{CE} + \alpha \sum_{l=1}^{M} \mathcal{L}^{(l)} \]
                \[ \mathcal{L}^{(l)} = -\sum_{c=1}^{C}\log p^{(l)}(c) \]
                \scriptsize
                % \textbf{Keys:} $\mathcal{L}_{CE}$ (CE Loss) $\cdot$ $\alpha$ (Weight) $\cdot$ $\tau$ (Temp)
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

		\begin{columns}[T]
			\begin{column}{0.50\textwidth}
				\begin{block}{Optimizer}
					\begin{itemize}
						\item \textbf{SGD, 30 epochs:} Stable convergence; training curves plateau before epoch 30.
						\item \textbf{LR = 0.001:} Standard for fine-tuning pretrained ResNet; avoids over-shooting.
						\item \textbf{Batch = 128:} Contrastive loss requires enough positive/negative pairs per step.
					\end{itemize}
				\end{block}
			\end{column}
			\begin{column}{0.50\textwidth}
				\begin{alertblock}{Architecture \& Loss}
    \begin{itemize}
        \item \textbf{$r \in \{2,4,6\}$:} Compression ratio of the projection module.
        \item \textbf{$\alpha \in [0.001,0.1]$:} Weight of the contrastive loss.
        \item \textbf{$\tau \in [0.8,1.4]$:} Temperature in contrastive learning.
    \end{itemize}
\end{alertblock}
			\end{column}
		\end{columns}

	\section{Conclusion}

	\begin{frame}{Summary of Key Contributions}
		\begin{block}{Core Achievements}
			\begin{itemize}
				\item \textbf{Architecture:} Combines multi-scale, multi-layer intermediate features while reducing reliance on domain-specific styles.
				\item \textbf{Loss Optimization:} Introduces a layer-wise contrastive loss that encourages class-level alignment across extraction levels.
				\item \textbf{Empirical Validation:} Shows competitive improvements over strong domain generalization baselines across three selected benchmark datasets.
			\end{itemize}
		\end{block}
	\end{frame}

	\begin{frame}{Limitations \& Future Directions}
		\begin{block}{Current Computational Drawbacks}
			\begin{itemize}
				\item \textbf{Memory Overhead:} Concatenating multiple intermediate feature maps creates dense hypercolumns before the classification head, increasing memory usage.
				\item \textbf{Batch Size Dependency:} Like most contrastive methods, $M^2$-CL requires large training batches to construct effective positive/negative pairs.
			\end{itemize}
		\end{block}
		\begin{alertblock}{Future Research Avenues}
			Explore cross-layer attention mechanisms, analyze object-level evidence more rigorously, and evaluate alternative similarity metrics like KL divergence to refine latent distribution shapes.
		\end{alertblock}
	\end{frame}

\begin{frame}{Baseline Methods Overview}
	\begin{columns}[T]
		\begin{column}{0.50\textwidth}
			\begin{block}{Statistical Alignment}
				\small
				\begin{itemize}
					\item \textbf{CORAL:} Aligns second-order statistics (covariance) across source domains.
					\item \textbf{MMD:} Minimizes Maximum Mean Discrepancy between domain distributions.
				\end{itemize}
			\end{block}
			\vspace{0.1cm}
			\begin{block}{Data Augmentation / Interpolation}
				\small
				\begin{itemize}
					\item \textbf{Mixup:} Interpolates images and labels across domain pairs.
					\item \textbf{SagNet:} Style-adversarial training to separate content from style.
				\end{itemize}
			\end{block}
		\end{column}
		\begin{column}{0.50\textwidth}
			\begin{alertblock}{Feature Regularization}
				\small
				\begin{itemize}
					\item \textbf{RSC:} Randomly suppresses dominant activations to prevent shortcut reliance.
					\item \textbf{SelfReg:} Self-regularization via contrastive and distance-based losses.
					\item \textbf{ARM:} Context-conditional adaptation at inference time.
					\item \textbf{SAGM:} Sharpness-aware gradient matching for flat loss minima.
					\item \textbf{EQRM:} Empirical quantile risk minimization to reduce tail risk.
				\end{itemize}
			\end{alertblock}
		\end{column}
	\end{columns}
\end{frame}
